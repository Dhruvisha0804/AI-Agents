Below are several sample user questions related to the MongoDB document provided, 
and the corresponding MongoDB aggregation pipeline queries that can be used to fetch the desired data.
Use them wisely.

Question 1. How many tasks are there by each different status?

Query
json
{
    "collection": "tasks", 
    "aggregate": [
        { "$group": { "_id": "$status.text", "count": { "$sum": 1 } } }
    ]
}


Question 2. How many tasks exist?

Query
json
{
    "collection": "tasks",
    "aggregate": [
        { "$count": "total_tasks" }
    ]
}


Question 3. List out all users who have task status 'done'

Query
json
{
    "collection": "tasks",
    "aggregate": [
        {
            "$lookup": {
                "from": "users",
                "localField": "AssigneeUserId",
                "foreignField": "_id",
                "as": "user_info"
            }
        },
        {
            "$unwind": "$user_info"
        },
        {
            "$match": {
                "status.text": "Done"
            }
        },
        {
            "$project": {
                "user_info.Employee_Name": 1,
                "status.text": 1,
                "_id": 0
            }
        }
    ]
}

Question 7. Filter out the employee whose '_id' ends with '68'.

Query
json
{
    "collection": "users",
    "filter": { 
        "$expr": { 
            "$regexMatch": { 
                "input": { "$toString": "$_id" }, 
                "regex": "68$" 
            } 
        } 
    },
    "projection": { "Employee_Name": 1, "_id": 1 }
}


Question 8. Aggregation to Count Tasks Created on a Specific Date.

Query
json
{
    "collection": "tasks",
    "aggregate": [
        {
            "$match": {
                "createdAt": {
                    "$gte": { "$date": "2023-01-01T00:00:00.000Z" },
                    "$lt": { "$date": "2024-01-01T00:00:00.000Z" }
                }
            }
        },
        {
            "$count": "total_tasks"
        }
    ]
}

Question 15. List all employees working on 'Alian Hr module' project.

Query
json
{
    "collection": "projects",
    "aggregate": [
        {
            "$match": {
                "ProjectName": "Alian Hr module"
            }
        },
        {
            "$lookup": {
                "from": "users",
                "let": {
                    "assignees": {
                        "$map": {
                            "input": "$AssigneeUserId",
                            "as": "id",
                            "in": {
                                "$toObjectId": "$$id"
                            }
                        }
                    }
                },
                "pipeline": [
                    {
                        "$match": {
                            "$expr": {
                                "$in": ["$_id", "$$assignees"]
                            }
                        }
                    },
                    {
                        "$project": {
                            "Employee_Name": 1,
                            "_id": 0
                        }
                    }
                ],
                "as": "employees"
            }
        },
        {
            "$unwind": "$employees"
        },
        {
            "$project": {
                "employees.Employee_Name": 1,
                "_id": 0
            }
        }
    ]
}

Question 20. Retrieve tasks under the project 'Alian Hub Design & Development' that have a 'MEDIUM' priority and were created last month.

                        "as": "project_info"
                    }
                }
            ]
        },
        "project_info.ProjectName": "Alian Hub Design & Development",
        "Task_Priority": "MEDIUM",
        "createdAt": {
            "$gte": { "$date": "2025-02-01T00:00:00.000Z" },
            "$lt": { "$date": "2025-03-01T00:00:00.000Z" }
        }
    },
    "projection": {
        "TaskName": 1,
        "Task_Priority": 1,
        "createdAt": 1,
        "_id": 0
    }
}
