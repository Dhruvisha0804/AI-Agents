List out all users who have task status 'done' 
{
    "collection": "tasks",
    "aggregate": [
        {
            "$lookup": {
                "from": "users",
                "localField": "AssigneeUserId",
                "foreignField": "_id",
                "as": "user_info"
            }
        },
        {
            "$unwind": "$user_info"
        },
        {
            "$match": {
                "status.text": "Done"
            }
        },
        {
            "$project": {
                "user_info.Employee_Name": 1,
                "status.text": 1,
                "_id": 0
            }
        }
    ]
}
